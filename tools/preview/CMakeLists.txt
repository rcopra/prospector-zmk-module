cmake_minimum_required(VERSION 3.16)
project(prospector_preview C)

set(ZMK_WORKSPACE "$ENV{HOME}/personal/zmk-config" CACHE PATH "West workspace supplying pinned LVGL")
unset(LV_CONF_PATH CACHE)
set(LV_CONF_INCLUDE_SIMPLE ON)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON)
set(LV_CONF_BUILD_DISABLE_DEMOS ON)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON)
set(VIEW "${CMAKE_CURRENT_SOURCE_DIR}/../../boards/shields/prospector_adapter/src/layouts/vaporwave")
add_subdirectory("${ZMK_WORKSPACE}/modules/lib/gui/lvgl" lvgl)
target_include_directories(lvgl SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(preview main.c ${VIEW}/view.c ${VIEW}/pixel_font.c ${VIEW}/status_font.c)
target_include_directories(preview PRIVATE ${VIEW})
target_link_libraries(preview PRIVATE lvgl)
enable_testing()
add_test(NAME vaporwave-layout COMMAND preview)

set(CATPPUCCIN "${VIEW}/../catppuccin")
add_executable(preview-catppuccin main.c ${CATPPUCCIN}/view.c ${VIEW}/pixel_font.c ${VIEW}/status_font.c)
target_include_directories(preview-catppuccin PRIVATE ${CATPPUCCIN})
target_compile_definitions(preview-catppuccin PRIVATE PREVIEW_CATPPUCCIN)
target_link_libraries(preview-catppuccin PRIVATE lvgl)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/catppuccin")
add_test(NAME catppuccin-layout COMMAND preview-catppuccin)
set_tests_properties(catppuccin-layout PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/catppuccin")
